public class ExtendedFinAnalysis { public static void main(String[] args) { // --- 1. System Parameters --- double tBase = 100.0; // Base Temperature (°C) double tAmbient = 25.0; // Ambient Air Temperature (°C) double length = 0.1; // Length L (m) = 10 cm double width = 0.05; // Width w (m) = 5 cm double thickness = 0.005; // Thickness t (m) = 0.5 cm double k = 205.0; // Conductivity of Aluminum (W/m·K) double h = 25.0; // Convective heat transfer coefficient (W/m²·K) // --- 2. Derived Geometric & Thermal Properties --- double perimeter = 2 * (width + thickness); double areaCross = width * thickness; double areaSurface = perimeter * length; double thetaBase = tBase - tAmbient; // Fin parameter: m = sqrt(h * P / (k * Ac)) double m = Math.sqrt((h * perimeter) / (k * areaCross)); // Base heat transfer constant: M = sqrt(h * P * k * Ac) * thetaBase double bigM = Math.sqrt(h * perimeter * k * areaCross) * thetaBase; // --- 3. Heat Transfer Performance Metrics (Insulated Tip) --- double qFin = bigM * Math.tanh(m * length); double qMax = h * areaSurface * thetaBase; double efficiency = (qFin / qMax) * 100.0; double effectiveness = qFin / (h * areaCross * thetaBase); // --- 4. Print Summary Diagnostics --- System.out.println("=================================================="); System.out.println(" RECTANGULAR FIN THERMAL ANALYSIS "); System.out.println("=================================================="); System.out.printf("Base Temp (Tb) : %.1f °C\n", tBase); System.out.printf("Ambient Temp (T_inf) : %.1f °C\n", tAmbient); System.out.printf("Fin Parameter (m) : %.4f m⁻¹\n", m); System.out.printf("Total Heat Rate (Q_fin): %.3f W\n", qFin); System.out.printf("Fin Efficiency (η_fin): %.2f %%\n", efficiency); System.out.printf("Fin Effectiveness (ε) : %.2f\n", effectiveness); System.out.println("--------------------------------------------------"); // --- 5. Temperature Profile Table Across 3 Boundary Conditions --- System.out.println("\n--- Temperature Profiles (°C) Along Fin Length ---"); System.out.printf("%-10s | %-16s | %-16s | %-16s\n", "x (m)", "1. Insulated Tip", "2. Convective Tip", "3. Infinite/Long"); System.out.println("----------------------------------------------------------------------"); int steps = 10; double dx = length / steps; for (int i = 0; i <= steps; i++) { double x = i * dx; // 1. Adiabatic / Insulated Tip double tInsulated = tAmbient + thetaBase * (Math.cosh(m * (length - x)) / Math.cosh(m * length)); // 2. Convective Tip (Heat lost from end face) double numConv = Math.cosh(m * (length - x)) + (h / (m * k)) * Math.sinh(m * (length - x)); double denConv = Math.cosh(m * length) + (h / (m * k)) * Math.sinh(m * length); double tConvective = tAmbient + thetaBase * (numConv / denConv); // 3. Very Long Fin (Infinitely long assumption: T -> T_inf) double tInfinite = tAmbient + thetaBase * Math.exp(-m * x); System.out.printf("%-10.3f | %-16.2f | %-16.2f | %-16.2f\n", x, tInsulated, tConvective, tInfinite); } System.out.println("======================================================================"); } }
